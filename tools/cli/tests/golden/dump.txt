== src/counter.vs (module counter)
-- Glow brush
shader Glow
uniform#0 viewport_size: Vec2F32
instance#0 rect_pos: Vec2F32
instance#1 rect_size: Vec2F32
instance#2 tint: ColorLinear
varying#0 local: Vec2F32

vertex(vertex_id: U32) -> VertexOutput builtins=[vertex_id]
  local unit: Vec2F32
  let unit = quad_vertex(vertex_id)
  varying.local = unit
  return VertexOutput(to_clip((instance.rect_pos + (unit * instance.rect_size)), uniform.viewport_size))

fragment() -> ColorLinear
  local k: F32
  let k = (1.0 - length((varying.local - Vec2F32(0.5, 0.5))))
  return ColorLinear((instance.tint.r * k), (instance.tint.g * k), (instance.tint.b * k), 1.0)
